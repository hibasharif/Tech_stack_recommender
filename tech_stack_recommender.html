# Step 2a: Build TF-IDF vectors from job role skill sets def build_tfidf_vectors(dataset, vocabulary): N = len(dataset) # Total number of job roles idf_scores = {} # Compute IDF for each skill in vocabulary for skill in vocabulary: df = sum(1 for role in dataset if skill in role['skills']) idf_scores[skill] = log(N / (df + 1)) # +1 smoothing # Build TF-IDF vector for each job role vectors = {} for role in dataset: tf_idf_vec = [] for skill in vocabulary: tf = role['skills'].count(skill) / len(role['skills']) val = tf * idf_scores[skill] tf_idf_vec.append(val) vectors[role['title']] = tf_idf_vec return vectors, idf_scores
# Step 2b: Cosine Similarity — measure angular alignment import math def cosine_similarity(vec_a, vec_b): """ cos(θ) = (A · B) / (||A|| × ||B||) Range: 0 to 1 (since TF-IDF values are non-negative) 1.0 = Perfect match | 0.0 = No overlap """ dot_product = sum(a * b for a, b in zip(vec_a, vec_b)) mag_a = math.sqrt(sum(a**2 for a in vec_a)) mag_b = math.sqrt(sum(b**2 for b in vec_b)) if mag_a == 0 or mag_b == 0: return 0.0 # Cold-start fallback return dot_product / (mag_a * mag_b)
# Full 4-Step Ranking Pipeline def recommend_careers(user_skills, dataset, top_n=3): # ── Step 1: INGESTION ──────────────────────────── vocabulary = sorted(set( skill for role in dataset for skill in role['skills'] )) user_vec = [1 if s in user_skills else 0 for s in vocabulary] # ── Step 2: SCORING ────────────────────────────── role_vectors, _ = build_tfidf_vectors(dataset, vocabulary) scores = {} for title, vec in role_vectors.items(): scores[title] = cosine_similarity(user_vec, vec) # ── Step 3: SORTING ────────────────────────────── ranked = sorted(scores.items(), key=lambda x: x[1], reverse=True) # ── Step 4: FILTERING (Top-N) ──────────────────── return ranked[:top_n]
